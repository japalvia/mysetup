Workaround is symlink:
sysroot=$(rustup run nightly-2026-10-01 rustc --print sysroot)
src="$sysroot/lib/rustlib/rustc-src/rust"
ln -fs ../../Cargo.lock "$src/compiler/rustc/Cargo.lock"
